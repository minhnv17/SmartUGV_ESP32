cmake_minimum_required(VERSION 3.16)
project(music_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CJSON "${CMAKE_CURRENT_SOURCE_DIR}/../managed_components/espressif__cjson/cJSON")
add_executable(music_catalog_test test_music_catalog.c ../main/nct_catalog.c ../main/display_text.c "${CJSON}/cJSON.c")
target_include_directories(music_catalog_test PRIVATE host ../main "${CJSON}")
if(MSVC)
    target_compile_options(music_catalog_test PRIVATE /utf-8)
else()
    target_link_libraries(music_catalog_test PRIVATE m)
endif()
enable_testing()
add_test(NAME music_catalog COMMAND music_catalog_test)
add_executable(speaker_test test_speaker.c ../main/speaker.c)
target_include_directories(speaker_test PRIVATE host ../main)
foreach(scenario RANGE 1 7)
    add_test(NAME speaker_${scenario} COMMAND speaker_test ${scenario})
endforeach()
add_executable(touch_test test_touch.c ../main/touch_ui.c)
target_include_directories(touch_test PRIVATE ../main)
add_test(NAME touch_controls COMMAND touch_test)
add_executable(mic_frontend_test test_mic_frontend.c ../main/mic_frontend.c)
target_include_directories(mic_frontend_test PRIVATE ../main)
if(NOT MSVC)
    target_link_libraries(mic_frontend_test PRIVATE m)
endif()
add_test(NAME microphone_frontend COMMAND mic_frontend_test)
